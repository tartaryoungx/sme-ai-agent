Token Monitoring · Optimization · Caching

3.1 — Token Tracking Schema

-- Supabase: token usage ต่อ shop และ session CREATE TABLE token_usage ( id uuid DEFAULT gen_random_uuid(), shop_id uuid NOT NULL, -- ร้านไหน session_id text, -- session ของลูกค้า model text, -- gemini-2.5-flash / haiku input_tokens integer DEFAULT 0, output_tokens integer DEFAULT 0, cached_tokens integer DEFAULT 0, -- tokens ที่ cache hit cost_usd numeric(10,8) DEFAULT 0, cache_hit boolean DEFAULT false, -- semantic cache hit? created_at timestamptz DEFAULT now() ); -- Monthly quota ต่อ shop (billing) CREATE TABLE shop_quota ( shop_id uuid PRIMARY KEY, plan text, -- starter/standard/pro monthly_tokens integer DEFAULT 3000000, -- 3M tokens/เดือน used_tokens integer DEFAULT 0, alert_at integer DEFAULT 80, -- alert เมื่อใช้ 80% reset_date date );

3.2 — 3-Layer Caching Strategy

CACHE LAYERS — ลำดับการเช็ก (ถูก → แพง)
LAYER 1
Semantic Cache (Redis + Vector)
เช็คว่าเคยมีคำถามคล้ายกัน (cosine similarity ≥ 0.92) → ตอบจาก cache เลย ไม่ยิง LLM
เหมาะกับ: FAQ, ถามราคา, เวลาเปิดร้าน, ที่อยู่
ประหยัด 100%
ไม่เสีย token เลย
↓ miss
LAYER 2
Prompt Cache (Gemini Context Cache)
Cache system prompt + knowledge base ของแต่ละร้าน ใน Gemini context cache
input ซ้ำๆ ลด $3.00 → $0.30/MTok (90% off)
TTL: 1 ชั่วโมง · auto-refresh ทุก session
ประหยัด ~60%
ของ input cost
↓ fresh call
LAYER 3
Context Compression
Sliding window: เก็บแค่ 6 turns ล่าสุด + summary ของ turns เก่า
ตัด redundant context ออกก่อนส่ง LLM
ลด input tokens 20-40% ต่อ request
ประหยัด 20-40%
ของ remaining
↓ LLM call with compressed context
POST-CALL
Token Logger → Langfuse → Supabase
บันทึก input/output/cached tokens, cost, latency, shop_id
update shop_quota.used_tokens · trigger alert ถ้า ≥ 80%

3.3 — Model Routing Logic

# model_router.py — AI Engineer #1 รับผิดชอบ async def route_model(message: str, context: dict) -> str: # STEP 1: เช็ค semantic cache ก่อน cached = await semantic_cache.lookup(message, shop_id=context["shop_id"]) if cached: return cached # ไม่เสีย token เลย # STEP 2: classify complexity is_simple = classify_intent(message) # Flash-Lite classifier # STEP 3: route ตาม complexity if is_simple: model = "gemini-2.5-flash-lite" # $0.10/$0.40 MTok # FAQ, เช็คเวลา, ถามราคาง่ายๆ else: model = "gemini-2.5-flash" # $0.30/$2.50 MTok # multi-step, tool-calling, complex CRM response = await call_llm( model=model, use_prompt_cache=True, # LAYER 2 เสมอ compressed_context=trim_context(context) # LAYER 3 ) # STEP 4: บันทึก usage await token_logger.record( shop_id=context["shop_id"], model=model, usage=response.usage, cost=calc_cost(response.usage, model) ) return response
